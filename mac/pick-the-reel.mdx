import { M7_1, M7_2, M7_3 } from "/snippets/video/m-reel.jsx";

Click **Reel** at the top. Pick **Game reel** or **Goalie reel**, then tick the moments you want.
You can zoom the replays, and take out talking near you.

<Steps>
  <Step title="Open Reel">
    Click **Reel** at the top.

    <M7_1 />
  </Step>
  <Step title="Pick the reel">
    Pick **Game reel** or **Goalie reel**.

    A game reel has the moments in order, with the score. A goalie reel has the saves.
  </Step>
  <Step title="Pick the moments">
    Tick In for each moment you want.
  </Step>
  <Step title="Slow it down">
    Tick Slow motion to replay a moment slowly.
  </Step>
  <Step title="Zoom the replays">
    Turn on **Zoom replays: saves on the goalie, goals on the whole shot**.

    <M7_2 />
  </Step>
  <Step title="Take out talking near you">
    Under Sound, pick **Remove nearby voices**.

    Or pick **Off** for no sound at all. A streamed game's reels start with Sound off.

    <M7_3 />
  </Step>
</Steps>
